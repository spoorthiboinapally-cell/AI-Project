<script>
/* ------------ BAD WORDS LIST ------------ */
const badWords = ["stupid","idiot","dumb","hate","fool","sucks","shut up","loser","dumbass","trash","garbage","worthless","dummy","moron"];

/* ------------ GENRES (unchanged) ------------ */
const genres = {
  "Fiction":["To Kill a Mockingbird","The Great Gatsby","Pride and Prejudice"],
  "Science Fiction":["Dune","The Martian","Ender's Game"],
  "Fantasy":["Harry Potter and the Sorcerer's Stone","The Hobbit","The Name of the Wind"]
  /* NOTE: Your full list was too long to paste here in this explanation.
     Use your original genres object — keep all items exactly as in your file. */
};

/* ------------ GLOBAL VARIABLES ------------ */
let userName = "";
let userSection = "";
let step = 0;
let lang = "en-IN";
let selectedGenre = "";
let selectedBook = "";
let borrowedBook = null;

/* ------------ FUNCTIONS ------------ */
function containsBadWords(text) {
  return badWords.some(word => text.toLowerCase().includes(word));
}

function speak(message) {
  const utter = new SpeechSynthesisUtterance(message);
  utter.lang = lang;
  speechSynthesis.speak(utter);
}

function addMessage(msg, sender, speakIt = true) {
  const chat = document.getElementById("chat");
  const div = document.createElement("div");
  div.className = "message " + sender;
  div.innerHTML = msg;
  chat.appendChild(div);
  chat.scrollTop = chat.scrollHeight;
  if (sender === "bot" && speakIt) speak(stripHTML(msg));
}

function stripHTML(text) {
  const div = document.createElement("div");
  div.innerHTML = text;
  return div.innerText;
}

function startListening() {
  const rec = new (window.SpeechRecognition || window.webkitSpeechRecognition)();
  rec.lang = lang;
  rec.start();
  addMessage("🎤 Listening...", "bot", false);

  rec.onresult = e => {
    const spoken = e.results[0][0].transcript;
    addMessage(spoken, "user", false);
    botReply(spoken);
  };
}

/* ------------ DATE HELPERS ------------ */
function getToday() {
  return new Date().toISOString().split("T")[0];
}
function getReturnDate() {
  const d = new Date();
  d.setDate(d.getDate() + 14);
  return d.toISOString().split("T")[0];
}

/* ------------ RESET ------------ */
function resetChat() {
  userName = "";
  userSection = "";
  step = 0;
  lang = document.getElementById("languageSelect").value;
  document.getElementById("chat").innerHTML = `
    <div class="message bot">Hello! I’m PLAB – Pragathi Library Assistant Bot.<br>What’s your name, reader?</div>
  `;
}

/* ------------ GENRE BUTTONS ------------ */
function showGenreButtons() {
  const box = document.getElementById("genreButtons");
  box.innerHTML = "";
  for (let genre in genres) {
    const btn = document.createElement("button");
    btn.textContent = genre;
    btn.onclick = () => botReply(genre);
    box.appendChild(btn);
  }
  box.style.display = "flex";
}

/* ------------ MAIN BOT LOGIC ------------ */
function sendMessage() {
  const input = document.getElementById("userInput");
  const text = input.value.trim();
  if (!text) return;
  addMessage(text, "user");
  input.value = "";
  botReply(text);
}

function botReply(text) {
  if (containsBadWords(text)) {
    addMessage("Please use polite language.", "bot");
    return;
  }

  text = text.trim();

  /* STEP 0 — Ask Name */
  if (step === 0) {
    userName = text;

    let msg = "";
    if (lang === "hi-IN") msg = `नमस्ते ${userName}! आप किस सेक्शन से हैं?`;
    else if (lang === "te-IN") msg = `హలో ${userName}! మీరు ఏ సెక్షన్ నుండి వచ్చారు?`;
    else msg = `Nice to meet you, ${userName}! Which section are you from?`;

    addMessage(msg, "bot");
    step = 1;
    return;
  }

  /* STEP 1 — Ask Section */
  if (step === 1) {
    userSection = text;

    let msg = "";
    if (lang === "hi-IN") msg = `बहुत बढ़िया, ${userName}! (${userSection} सेक्शन)  
कृपया नीचे दिए गए जेनर में से एक चुनें।`;
    else if (lang === "te-IN") msg = `బాగుంది, ${userName}! (${userSection} సెక్షన్)  
దయచేసి క్రింద ఉన్న జానర్ బటన్లలో ఒకదాన్ని ఎంచుకోండి.`;
    else msg = `Great, ${userName}! (Section: ${userSection})  
Please choose your favorite genre from the buttons below.`;

    addMessage(msg, "bot");
    showGenreButtons();
    step = 2;
    return;
  }

  /* STEP 2 — Choose Genre */
  if (step === 2) {
    const genreNames = Object.keys(genres).map(g => g.toLowerCase());
    if (!genreNames.includes(text.toLowerCase())) {
      addMessage("Please choose a genre from the buttons.", "bot");
      showGenreButtons();
      return;
    }

    selectedGenre = Object.keys(genres).find(g => g.toLowerCase() === text.toLowerCase());
    const books = genres[selectedGenre];
    const html = books.map(b => `<li>${b}</li>`).join("");

    let msg = "";
    if (lang === "hi-IN") msg = `बहुत बढ़िया! ${selectedGenre} की पुस्तकें:<ul class="book-list">${html}</ul> कृपया पुस्तक का नाम टाइप करें।`;
    else if (lang === "te-IN") msg = `చాలా బాగుంది! ${selectedGenre} పుస్తకాల జాబితా:<ul class="book-list">${html}</ul> దయచేసి పుస్తకాన్ని టైప్ చేయండి.`;
    else msg = `Great choice! Here are some ${selectedGenre} books:<ul class="book-list">${html}</ul>Please type the book name you want.`;

    addMessage(msg, "bot");
    document.getElementById("genreButtons").style.display = "none";
    step = 3;
    return;
  }

  /* STEP 3 — Choose Book */
  if (step === 3) {
    const lowerList = genres[selectedGenre].map(b => b.toLowerCase());
    if (!lowerList.includes(text.toLowerCase())) {
      addMessage("That book is not in the list. Try again.", "bot");
      return;
    }

    borrowedBook = text;
    const bdate = getToday();
    const rdate = getReturnDate();

    let msg = "";
    if (lang === "hi-IN") msg = `"${borrowedBook}" आपको ${bdate} को दी गई है। कृपया इसे ${rdate} तक लौटा दें।`;
    else if (lang === "te-IN") msg = `"${borrowedBook}" పుస్తకం మీకు ${bdate} న ఇవ్వబడింది. దయచేసి ${rdate} నాటికి తిరిగి ఇవ్వండి.`;
    else msg = `You borrowed "${borrowedBook}" on ${bdate}. Please return it by ${rdate}.`;

    addMessage(msg, "bot");
    step = 4;
    return;
  }

  /* STEP 4 — User already borrowed */
  if (step === 4) {
    addMessage(`You already borrowed "${borrowedBook}". Return it before borrowing another.`, "bot");
  }
}
