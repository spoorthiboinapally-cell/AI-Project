print("♻️ SMART WASTE SEGREGATION BIN ♻️")
print("Bins: Recyclable 🟦 | Biodegradable 🟩 | Non-Biodegradable 🟥\n")
print("Type 'status' to view bin levels, 'empty' to clear bins, or 'exit' to stop.\n")

bins = {
    "Recyclable Bin 🟦": {"count": 0, "capacity": 10},
    "Biodegradable Bin 🟩": {"count": 0, "capacity": 10},
    "Non-Biodegradable Bin 🟥": {"count": 0, "capacity": 10},
}

def progress_bar(current, total, length=20):
    filled = int(length * current / total)
    bar = '█' * filled + '-' * (length - filled)
    percent = (current / total) * 100
    return f"[{bar}] {percent:.0f}%"


def show_status():
    print("\n📊 BIN STATUS:")
    for name, data in bins.items():
        bar = progress_bar(data['count'], data['capacity'])
        status = "✅ OK" if data["count"] < data["capacity"] else "❌ FULL"
        print(f"{name}: {data['count']}/{data['capacity']} {bar} {status}")
    print()


def empty_bins():
    for b in bins.values():
        b["count"] = 0
    print("\n🧹 All bins emptied successfully!\n")


# ---------------------------------------------------------
# MAIN LOOP (NOW SUPPORTS MULTIPLE ITEMS!)
# ---------------------------------------------------------

while True:
    item = input("Enter waste item (or 'status', 'empty', 'exit'): ").strip()

    if item.lower() == 'exit':
        print("Segregation system shutting down. Goodbye! 👋")
        break

    elif item.lower() == 'status':
        show_status()
        continue

    elif item.lower() == 'empty':
        empty_bins()
        continue

    # ⭐ Detect quantity
    parts = item.split()
    qty = 1

    if parts[0].isdigit():
        qty = int(parts[0])
        item_name = " ".join(parts[1:])
    else:
        item_name = item

    result = segregate_waste(item_name)

    # Unknown item message
    if result == "Unknown":
        print(f"🙂 '{item_name}' is not on the list for now. Please wait — we will try to add it within 24 hours.\n")
        continue

    available_space = bins[result]["capacity"] - bins[result]["count"]

    if available_space <= 0:
        print(f"🚫 {result} is FULL! Please empty it before adding more.\n")
        continue

    # ⭐ If user tries to add more than available
    if qty > available_space:
        bins[result]["count"] += available_space
        print(f"⚠️ Only {available_space} of {qty} '{item_name}' added. {result} is now FULL!\n")
        continue

    # Normal add
    bins[result]["count"] += qty
    print(f"✅ Added {qty} '{item_name}' → {result}")

    if bins[result]["count"] == bins[result]["capacity"]:
        print(f"⚠️ {result} is now FULL!\n")
    else:
        remaining = bins[result]['capacity'] - bins[result]['count']
        print(f"🪣 Current load: {bins[result]['count']}/{bins[result]['capacity']} ({remaining} spaces left)\n")
